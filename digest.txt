"use strict";

// ---------- WPM logic (pure, no DOM, no input handling) ----------
function calcWpm(charCount, elapsedMs) {
  if (!(elapsedMs > 0) || charCount <= 0) return 0; // no division by zero
  return (charCount / 5) / (elapsedMs / 60000);
}

function calcAccuracy(correctCount, totalCount) {
  if (totalCount <= 0) return 0;
  return (correctCount / totalCount) * 100;
}

// ---------- Config / state ----------
const DEFAULT_TEXT = "The quick brown fox jumps over the lazy dog. Pack my box with five dozen liquor jugs, and sphinx of black quartz, judge my vow!";
const MIN_DURATION = 10;
const MAX_DURATION = 30;

const state = {
  target: DEFAULT_TEXT,   // active target; only changed by applyText()
  typed: [],              // typed characters
  duration: 30,           // seconds
  status: "idle",         // idle | running | done
  startTime: 0,
  elapsedMs: 0,
  intervalId: null,
};

// ---------- DOM ----------
const $target = document.getElementById("target");
const $wpm = document.getElementById("wpm");
const $wpmLabel = document.getElementById("wpm-label");
const $timer = document.getElementById("timer");
const $custom = document.getElementById("custom");
const $duration = document.getElementById("duration");
const $results = document.getElementById("results");
const $resWpm = document.getElementById("res-wpm");
const $resAcc = document.getElementById("res-acc");
const $restart = document.getElementById("restart");
const $apply = document.getElementById("apply");

let spans = [];

// ---------- Rendering ----------
function buildTarget() {
  $target.textContent = "";
  spans = [...state.target].map(ch => {
    const s = document.createElement("span");
    s.textContent = ch;
    $target.appendChild(s);
    return s;
  });
}

function render() {
  const t = [...state.target];
  for (let i = 0; i < spans.length; i++) {
    let cls = "pending";
    if (i < state.typed.length) cls = state.typed[i] === t[i] ? "correct" : "incorrect";
    else if (i === state.typed.length && state.status !== "done") cls = "current";
    spans[i].className = cls;
  }
  const elapsed = state.status === "running" ? performance.now() - state.startTime : state.elapsedMs;
  $wpm.textContent = Math.round(calcWpm(state.typed.length, elapsed));
  $wpmLabel.textContent = state.status === "done" ? "Final WPM" : "WPM";
  const remaining = Math.max(0, state.duration * 1000 - elapsed);
  $timer.textContent = (remaining / 1000).toFixed(1) + "s";
}

function showResults() {
  // Based on the final typed state only; corrected mistakes are already gone.
  const t = [...state.target];
  const correct = state.typed.reduce((n, ch, i) => n + (ch === t[i] ? 1 : 0), 0);
  $resWpm.textContent = calcWpm(state.typed.length, state.elapsedMs).toFixed(1);
  $resAcc.textContent = calcAccuracy(correct, state.typed.length).toFixed(1);
  $results.hidden = false;
}

// ---------- Test lifecycle ----------
function resetTest() {
  clearInterval(state.intervalId);
  state.typed = [];
  state.status = "idle";
  state.startTime = 0;
  state.elapsedMs = 0;
  $results.hidden = true;
  buildTarget();
  render();
}

function startTimer() {
  state.status = "running";
  state.startTime = performance.now();
  state.intervalId = setInterval(tick, 100);
}

function finish(elapsedMs) {
  clearInterval(state.intervalId);
  state.status = "done";
  state.elapsedMs = elapsedMs;
  render();
  showResults();
}

function tick() {
  const elapsed = performance.now() - state.startTime;
  const limit = state.duration * 1000;
  if (elapsed >= limit) finish(limit); // clamp to exact limit
  else render();
}

// ---------- Keyboard input ----------
document.addEventListener("keydown", (e) => {
  if (e.target.closest("textarea, input, select")) return; // don't hijack form fields
  if (e.isComposing || e.ctrlKey || e.metaKey || e.altKey) return;
  if (state.status === "done") return;

  if (e.key === "Backspace") {
    e.preventDefault();
    if (state.typed.length > 0) state.typed.pop();
    render();
    return;
  }

  if (e.key.length !== 1) return; // Shift, Ctrl, arrows, Tab, Enter, etc.
  e.preventDefault();             // stops space from scrolling

  if (state.typed.length >= state.target.length) return;
  if (state.status === "idle") startTimer();

  state.typed.push(e.key);

  if (state.typed.length === state.target.length) {
    finish(performance.now() - state.startTime);
  } else {
    render();
  }
});

// ---------- Controls ----------
function applyText() {
  // Newlines/tabs can't be typed, collapse all whitespace to single spaces.
  const cleaned = $custom.value.replace(/\s+/g, " ").trim();
  if (!cleaned) return; // empty/whitespace only: keep current test untouched
  state.target = cleaned;
  resetTest();
}

function applyDuration() {
  let v = Math.round(Number($duration.value));
  if (!Number.isFinite(v) || $duration.value === "") v = state.duration;
  v = Math.min(MAX_DURATION, Math.max(MIN_DURATION, v));
  $duration.value = v;
  state.duration = v;
  resetTest();
}

$restart.addEventListener("click", () => { resetTest(); $restart.blur(); });
$apply.addEventListener("click", () => { applyText(); $apply.blur(); });
$duration.addEventListener("change", () => { applyDuration(); $duration.blur(); });

// ---------- Init ----------
$custom.value = DEFAULT_TEXT;
resetTest();
